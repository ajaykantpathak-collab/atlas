// HELPER FUNCTIONS

function getMasteryKey(nodeId: string): string {
  return `${STORAGE_KEY}${nodeId}`
}

function loadMastery(nodeId: string): MasteryRecord | null {
  if (typeof window === 'undefined') return null
  try {
    const raw = localStorage.getItem(getMasteryKey(nodeId))
    return raw ? JSON.parse(raw) : null
  } catch {
    return null
  }
}

function saveMastery(nodeId: string, record: MasteryRecord): void {
  if (typeof window === 'undefined') return
  try {
    localStorage.setItem(getMasteryKey(nodeId), JSON.stringify(record))
  } catch {
    // quota exceeded silently
  }
}

function isReviewDue(lastReview: string, nextReview: string): boolean {
  return new Date(nextReview) <= new Date()
}

function calculateNextReview(attempts: number): string {
  const idx = Math.min(attempts, REVIEW_INTERVAL_DAYS.length - 1)
  const d = new Date()
  d.setDate(d.getDate() + REVIEW_INTERVAL_DAYS[idx])
  return d.toISOString()
}

function getElementByZ(z: number): ElementDatum | undefined {
  return ELEMENTS.find((e) => e.z === z)
}

/** Interactive periodic table showing groups and periods */
const PeriodicTableSVG: React.FC<{
  onSelect: (el: ElementDatum) => void
  selectedZ: number | null
  highlightCategory?: string
  width?: number
}> = ({ onSelect, selectedZ, highlightCategory, width = 850 }) => {
  const cellW = 44
  const cellH = 44
  const gap = 3
  const ox = 15
  const oy = 40

  return (
    <svg viewBox="0 0 900 360" className="w-full max-w-4xl mx-auto" style={{ maxWidth: width }}>
      {/* Group labels */}
      <text x={ox + 0.5 * cellW} y={20} textAnchor="middle" fontSize="10" fill="#666">1</text>
      <text x={ox + 2.5 * (cellW + gap)} y={20} textAnchor="middle" fontSize="10" fill="#666">2</text>
      <text x={ox + 12.5 * (cellW + gap)} y={20} textAnchor="middle" fontSize="10" fill="#666">13</text>
      <text x={ox + 17.5 * (cellW + gap)} y={20} textAnchor="middle" fontSize="10" fill="#666">18</text>
      <text x={5} y={oy + 0.5 * cellH} textAnchor="middle" fontSize="10" fill="#666">1</text>
      <text x={5} y={oy + 3.5 * cellH + 3 * gap} textAnchor="middle" fontSize="10" fill="#666">2</text>
      <text x={5} y={oy + 4.5 * cellH + 4 * gap} textAnchor="middle" fontSize="10" fill="#666">3</text>
      <text x={5} y={oy + 5.5 * cellH + 5 * gap} textAnchor="middle" fontSize="10" fill="#666">4</text>

      {/* Render elements */}
      {ELEMENTS.map((el) => {
        const x = ox + (el.col - 1) * (cellW + gap)
        const y = oy + (el.row - 1) * (cellH + gap)
        const isSelected = selectedZ === el.z
        const dimmed = highlightCategory && el.category !== highlightCategory
        return (
          <g
            key={el.z}
            onClick={() => onSelect(el)}
            className="cursor-pointer"
            style={{ opacity: dimmed ? 0.2 : 1 }}
          >
            <rect
              x={x}
              y={y}
              width={cellW}
              height={cellH}
              rx={4}
              fill={CATEGORY_COLORS[el.category] || '#999'}
              stroke={isSelected ? '#fff' : 'none'}
              strokeWidth={isSelected ? 2.5 : 0}
            />
            <text x={x + cellW / 2} y={y + cellH / 2 - 2} textAnchor="middle" fontSize="13" fontWeight="bold" fill="#fff">
              {el.symbol}
            </text>
            <text x={x + cellW / 2} y={y + cellH / 2 + 12} textAnchor="middle" fontSize="8" fill="#fff">
              {el.z}
            </text>
          </g>
        )
      })}

      {/* Legend */}
      <g transform="translate(15, 290)">
        {Object.entries(CATEGORY_COLORS).map(([cat, color]) => (
          <g key={cat} transform={`translate(${Object.keys(CATEGORY_COLORS).indexOf(cat) * 90}, 0)`}>
            <rect width={16} height={14} rx={2} fill={color} />
            <text x={20} y={11} fontSize="9" fill="#333">
              {cat.replace('-', ' ')}
            </text>
          </g>
        ))}
      </g>
    </svg>
  )
}

            </marker>
          </defs>
          <line x1={ax + 30} y1={cy + 15} x2={cx - 10} y2={cy + 5} stroke="#e67e22" strokeWidth={2} markerEnd="url(#coordArrow)" strokeDasharray="4,2" />
        </>
      )}

      {/* Labels */}
      <text x={cx} y={330} textAnchor="middle" fontSize="16" fontWeight="bold" fill="#2c3e50">
        {bondType === 'ionic' ? 'Ionic Bond' : bondType === 'covalent' ? 'Covalent Bond' : 'Coordinate Bond'}
      </text>
      <text x={cx} y={355} textAnchor="middle" fontSize="12" fill="#666">
        {bondType === 'ionic'

      </text>
    </svg>
  )
}

/** Titration curve SVG */
const TitrationCurveSVG: React.FC<{
  volume: number
  equivalenceVol: number
  acidType: 'strong' | 'weak'
  baseType: 'strong' | 'weak'
}> = ({ volume, equivalenceVol, acidType, baseType }) => {
  const w = 700
  const h = 400
  const padX = 60
  const padY = 40
  const plotW = w - 2 * padX
  const plotH = h - 2 * padY
  const maxVol = 50

  const isStrongStrong = acidType === 'strong' && baseType === 'strong'
  const startPH = acidType === 'strong' ? 1 : 3
  const endPH = baseType === 'strong' ? 13 : 11
  const eqPH = isStrongStrong ? 7 : 7
  const jumpHeight = isStrongStrong ? 10 : 4
  const eqY = padY + plotH - ((eqPH - 0) / 14) * plotH
  const startY = padY + plotH - ((startPH - 0) / 14) * plotH
  const endY = padY + plotH - ((endPH - 0) / 14) * plotH
  const volX = padX + (volume / maxVol) * plotW
  const eqX = padX + (equivalenceVol / maxVol) * plotW

  // Build the S-curve path
  const points: string[] = []
  const steps = 80
  for (let i = 0; i <= steps; i++) {
    const v = (i / steps) * maxVol
    const x = padX + (v / maxVol) * plotW
    let ph: number
    if (v < equivalenceVol - 2) {
      ph = startPH + (v / (equivalenceVol - 2)) * (eqPH + jumpHeight - startPH) * 0.3
    } else if (v < equivalenceVol + 2) {
      const t = (v - (equivalenceVol - 2)) / 4
      ph = startPH + 0.3 * (eqPH + jumpHeight - startPH) + t * jumpHeight * 2
    } else {
      const t = Math.min((v - equivalenceVol - 2) / 10, 1)
      ph = eqPH + jumpHeight * 2 + t * (endPH - eqPH - jumpHeight * 2)
    }
    ph = Math.max(0, Math.min(14, ph))
    const y = padY + plotH - (ph / 14) * plotH
    points.push(`${i === 0 ? 'M' : 'L'}${x.toFixed(1)},${y.toFixed(1)}`)
  }

  return (
    <svg viewBox={`0 0 ${w} ${h}`} className="w-full max-w-2xl mx-auto">
      {/* Axes */}
      <line x1={padX} y1={padY} x2={padX} y2={padY + plotH} stroke="#333" strokeWidth={1.5} />
      <line x1={padX} y1={padY + plotH} x2={padX + plotW} y2={padY + plotH} stroke="#333" strokeWidth={1.5} />

      {/* Arrows */}
      <polygon points={`${padX},${padY - 5} ${padX - 5},${padY + 5} ${padX + 5},${padY + 5}`} fill="#333" />
      <polygon points={`${padX + plotW + 5},${padY + plotH} ${padX + plotW - 5},${padY + plotH - 5} ${padX + plotW - 5},${padY + plotH + 5}`} fill="#333" />

      {/* Labels */}
      <text x={padX / 2} y={padY + plotH / 2} textAnchor="middle" fontSize="12" fill="#333" transform={`rotate(-90, ${padX / 2}, ${padY + plotH / 2})`}>pH</text>
      <text x={padX + plotW / 2} y={h - 5} textAnchor="middle" fontSize="12" fill="#333">Volume of titrant (mL)</text>

      {/* Tick marks */}
      {[0, 7, 14].map((ph) => (
        <g key={ph}>
          <line x1={padX - 5} y1={padY + plotH - (ph / 14) * plotH} x2={padX} y2={padY + plotH - (ph / 14) * plotH} stroke="#666" />
          <text x={padX - 10} y={padY + plotH - (ph / 14) * plotH + 4} textAnchor="end" fontSize="10" fill="#666">{ph}</text>
        </g>
      ))}
      {[0, 25, 50].map((vol) => (
        <g key={vol}>
          <line x1={padX + (vol / maxVol) * plotW} y1={padY + plotH} x2={padX + (vol / maxVol) * plotW} y2={padY + plotH + 5} stroke="#666" />
          <text x={padX + (vol / maxVol) * plotW} y={padY + plotH + 18} textAnchor="middle" fontSize="10" fill="#666">{vol}</text>
        </g>
      ))}

      {/* Equivalence point line */}
      <line x1={eqX} y1={padY} x2={eqX} y2={padY + plotH} stroke="#e74c3c" strokeDasharray="4,4" opacity={0.5} />

      {/* Titration curve */}
      <path d={points.join(' ')} fill="none" stroke="#2980b9" strokeWidth={2.5} />

      {/* Current point */}
      {volume > 0 && (
        <circle cx={volX} cy={padY + plotH - ((Math.min(startPH + (volume / maxVol) * 8, 14) - 0) / 14) * plotH} r={6} fill="#e74c3c" />
      )}

      {/* Equivalence label */}
      <text x={eqX + 5} y={eqY - 8} fontSize="11" fill="#e74c3c" fontWeight="bold">Equivalence Point</text>

      {/* Info */}
      <text x={w / 2} y={15} textAnchor="middle" fontSize="12" fill="#333">
        {acidType === 'strong' && baseType === 'strong'
          ? 'Strong Acid → Strong Base'
          : acidType === 'weak' && baseType === 'strong'
            ? 'Weak Acid → Strong Base'
            : 'Strong Acid → Weak Base'}
      </text>
    </svg>
  )
}

const MoleCalculatorLab: React.FC = () => {
  const [mass, setMass] = useState('')
  const [molarMass, setMolarMass] = useState('')
  const [result, setResult] = useState<string | null>(null)

  const calculate = useCallback(() => {
    const m = parseFloat(mass)
    const M = parseFloat(molarMass)
    if (!isNaN(m) && !isNaN(M) && M > 0) {
      const moles = m / M
      setResult(`${moles.toFixed(4)} mol`)
    } else {
      setResult('Enter valid numbers')
    }
  }, [mass, molarMass])

  return (
    <div className="bg-white rounded-lg p-6 shadow max-w-md mx-auto space-y-4">
      <h4 className="font-bold text-center text-lg">Mole Calculator</h4>
      <div>
        <label className="block text-sm font-medium mb-1">Mass (g)</label>
        <input type="number" value={mass} onChange={(e) => setMass(e.target.value)} className="w-full border rounded px-3 py-2" placeholder="e.g., 117" />
      </div>
      <div>
        <label className="block text-sm font-medium mb-1">Molar Mass (g/mol)</label>
        <input type="number" value={molarMass} onChange={(e) => setMolarMass(e.target.value)} className="w-full border rounded px-3 py-2" placeholder="e.g., 58.5" />
      </div>
      <button onClick={calculate} className="w-full bg-blue-600 text-white rounded py-2 font-medium hover:bg-blue-700 transition">
        Calculate Moles
      </button>
      {result && <p className="text-center text-lg font-bold text-blue-700">Moles = {result}</p>}
      <p className="text-xs text-gray-500 text-center">Formula: n = mass / Molar Mass</p>
    </div>
  )
}

const ChallengeRenderer: React.FC<{
  challenge: ChallengeItem
  onAnswer: (answer: string, isCorrect: boolean) => void
  hintIndex: number
  onRequestHint: () => void
  showFeedback: boolean
  isCorrect: boolean
  misconception: string | null
  userAnswer: string
  onAnswerChange: (val: string) => void
}> = ({ challenge, onAnswer, hintIndex, onRequestHint, showFeedback, isCorrect, misconception, userAnswer, onAnswerChange }) => {
  const handleSubmit = useCallback(() => {
    if (!userAnswer.trim()) return
    const correct = userAnswer.trim().toLowerCase() === challenge.correctAnswer.toLowerCase()
    onAnswer(userAnswer.trim(), correct)
  }, [userAnswer, challenge.correctAnswer, onAnswer])

  const handleOptionSelect = useCallback(
    (option: string) => {
      const correct = option.toLowerCase() === challenge.correctAnswer.toLowerCase()
      onAnswer(option, correct)
    },
    [challenge.correctAnswer, onAnswer]
  )

  return (
    <motion.div initial={{ opacity: 0, x: 20 }} animate={{ opacity: 1, x: 0 }} className="bg-white rounded-xl p-6 shadow-lg max-w-2xl mx-auto">
      <div className="flex items-center gap-2 mb-3">
        <span className="px-2 py-1 rounded-full text-xs font-medium bg-blue-100 text-blue-700">{challenge.topic}</span>
        <span className={`px-2 py-1 rounded-full text-xs font-medium ${challenge.difficulty === 'foundation' ? 'bg-green-100 text-green-700' : challenge.difficulty === 'standard' ? 'bg-yellow-100 text-yellow-700' : 'bg-red-100 text-red-700'}`}>
          {challenge.difficulty}
        </span>
        <span className="px-2 py-1 rounded-full text-xs font-medium bg-purple-100 text-purple-700">{challenge.bloomLevel}</span>
      </div>

      <h3 className="text-lg font-semibold mb-4 text-gray-800">{challenge.question}</h3>

      {challenge.formula && !showFeedback && <p className="text-xs text-gray-500 mb-3 font-mono bg-gray-50 p-2 rounded whitespace-pre-line">{challenge.formula}</p>}

      {challenge.answerType === 'mcq' && challenge.options && !showFeedback && (
        <div className="grid grid-cols-2 gap-3 mb-4">
          {challenge.options.map((opt) => (
            <button key={opt} onClick={() => handleOptionSelect(opt)} className="px-4 py-3 border-2 border-gray-200 rounded-lg hover:border-blue-400 hover:bg-blue-50 transition text-left text-sm font-medium">
              {opt}
            </button>
          ))}
        </div>
      )}

      {challenge.answerType === 'number' && !showFeedback && (
        <div className="flex gap-3 mb-4 items-center">
          <input
            type="number"
            value={userAnswer}
            onChange={(e) => onAnswerChange(e.target.value)}
            onKeyDown={(e) => e.key === 'Enter' && handleSubmit()}
            className="flex-1 border-2 border-gray-300 rounded-lg px-4 py-2 text-lg focus:border-blue-500 focus:outline-none"
            placeholder="Enter your answer..."
            autoFocus
          />
          {challenge.unit && <span className="text-sm text-gray-500 font-medium">{challenge.unit}</span>}
          <button onClick={handleSubmit} disabled={!userAnswer.trim()} className="px-6 py-2 bg-blue-600 text-white rounded-lg font-medium disabled:opacity-50 hover:bg-blue-700 transition">
            Submit
          </button>
        </div>
      )}

      {/* Hints */}
      <AnimatePresence>
        {hintIndex > 0 && !showFeedback && (
          <motion.div initial={{ opacity: 0, height: 0 }} animate={{ opacity: 1, height: 'auto' }} className="mb-3">
            {challenge.hints.slice(0, hintIndex).map((hint, i) => (
              <div key={i} className="flex items-start gap-2 bg-yellow-50 border border-yellow-200 rounded-lg p-2 mt-2">
                <span className="text-yellow-500 mt-0.5">💡</span>
                <p className="text-sm text-yellow-800">{hint}</p>
              </div>
            ))}
          </motion.div>
        )}
      </AnimatePresence>

      {!showFeedback && hintIndex < challenge.hints.length && (
        <button onClick={onRequestHint} className="text-sm text-blue-600 hover:text-blue-800 underline">
          💡 Show Hint {hintIndex + 1}/{challenge.hints.length}
        </button>
      )}

      {/* Feedback */}
      <AnimatePresence>
        {showFeedback && (
          <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} className="mt-4">
            {isCorrect ? (
              <div className="bg-green-50 border border-green-300 rounded-lg p-4">
                <p className="text-green-700 font-semibold text-lg">✅ Correct!</p>
                {challenge.working && <p className="text-sm text-green-600 mt-1 whitespace-pre-line">{challenge.working}</p>}
              </div>
            ) : (
              <div className="bg-red-50 border border-red-300 rounded-lg p-4">
                <p className="text-red-700 font-semibold text-lg">❌ Incorrect</p>
                <div className="bg-red-100 rounded p-3 mt-2">
                  <p className="text-sm font-semibold text-red-600">Common Misconception:</p>
                  <p className="text-sm text-red-500 italic">"{misconception}"</p>
                  <p className="text-sm text-red-700 mt-1"><strong>Correct answer:</strong> {challenge.correctAnswer}{challenge.unit ? ` ${challenge.unit}` : ''}</p>
                </div>
                {challenge.working && <p className="text-sm text-gray-600 mt-2 whitespace-pre-line">{challenge.working}</p>}
              </div>
            )}
          </motion.div>
        )}
      </AnimatePresence>
    </motion.div>
  )
}

      setHintIndex(0)

      setCurrentMisconception(null)
    } else {

    }

    setCurrentMisconception(null)

    }

function getMasteryKey(nodeId: string): string {
  return `${STORAGE_KEY}${nodeId}`
}

function loadMastery(nodeId: string): MasteryRecord | null {
  if (typeof window === 'undefined') return null
  try {
    const raw = localStorage.getItem(getMasteryKey(nodeId))
    return raw ? JSON.parse(raw) : null
  } catch {
    return null
  }
}

function saveMastery(nodeId: string, record: MasteryRecord): void {
  if (typeof window === 'undefined') return
  try {
    localStorage.setItem(getMasteryKey(nodeId), JSON.stringify(record))
  } catch {
    // quota exceeded
  }
}

function isReviewDue(lastReview: string, nextReview: string): boolean {
  return new Date(nextReview) <= new Date()
}

function calculateNextReview(attempts: number): string {
  const idx = Math.min(attempts, REVIEW_INTERVAL_DAYS.length - 1)
  const d = new Date()
  d.setDate(d.getDate() + REVIEW_INTERVAL_DAYS[idx])
  return d.toISOString()
}

function formatCurrency(amount: number): string {
  return '₹' + amount.toLocaleString('en-IN')
}

/** Lab 5: Depreciation Calculator */
const DepreciationCalculator: React.FC = () => {
  const [cost, setCost] = useState(100000)
  const [scrap, setScrap] = useState(10000)
  const [life, setLife] = useState(5)
  const [method, setMethod] = useState<'SLM' | 'WDV'>('SLM')
  const [wdvRate, setWdvRate] = useState(20)

  const slmDep = (cost - scrap) / life
  const wdvDep = method === 'WDV' ? (cost - scrap) * (wdvRate / 100) : 0

  const years = Array.from({ length: life }, (_, i) => i + 1)
  const rows = years.map((yr) => {
    if (method === 'SLM') {
      const bookValue = cost - slmDep * yr
      return { year: yr, depreciation: slmDep, bookValue: Math.max(bookValue, scrap) }
    }
    const prevBV = yr === 1 ? cost : 0
    let bv = cost
    for (let y = 1; y <= yr; y++) {
      bv = bv * (1 - wdvRate / 100)
    }
    bv = Math.max(bv, scrap)
    return { year: yr, depreciation: bv < scrap ? 0 : (method === 'WDV' ? 0 : 0), bookValue: bv }
  })

  return (
    <div className="bg-white rounded-xl p-5 shadow max-w-md mx-auto space-y-4">
      <h4 className="font-bold text-center">Depreciation Calculator</h4>

      <div>
        <label className="block text-sm font-medium mb-1">Method</label>
        <div className="flex gap-2">
          <button onClick={() => setMethod('SLM')} className={`flex-1 py-1.5 rounded text-sm ${method === 'SLM' ? 'bg-purple-600 text-white' : 'bg-gray-100'}`}>Straight Line (SLM)</button>
          <button onClick={() => setMethod('WDV')} className={`flex-1 py-1.5 rounded text-sm ${method === 'WDV' ? 'bg-purple-600 text-white' : 'bg-gray-100'}`}>Written Down (WDV)</button>
        </div>
      </div>

      <div className="grid grid-cols-2 gap-3">
        <div>
          <label className="block text-xs text-gray-500 mb-1">Cost (₹)</label>
          <input type="number" value={cost} onChange={(e) => setCost(Number(e.target.value))} className="w-full border rounded px-2 py-1 text-sm" />
        </div>
        <div>
          <label className="block text-xs text-gray-500 mb-1">Scrap Value (₹)</label>
          <input type="number" value={scrap} onChange={(e) => setScrap(Number(e.target.value))} className="w-full border rounded px-2 py-1 text-sm" />
        </div>
        <div>
          <label className="block text-xs text-gray-500 mb-1">Useful Life (years)</label>
          <input type="number" value={life} onChange={(e) => setLife(Number(e.target.value))} className="w-full border rounded px-2 py-1 text-sm" />
        </div>
        {method === 'WDV' && (
          <div>
            <label className="block text-xs text-gray-500 mb-1">WDV Rate (%)</label>
            <input type="number" value={wdvRate} onChange={(e) => setWdvRate(Number(e.target.value))} className="w-full border rounded px-2 py-1 text-sm" />
          </div>
        )}
      </div>

      <div className="bg-purple-50 rounded p-3 text-center">
        <p className="text-sm text-purple-700">
          Annual Depreciation ({method}): <span className="font-bold text-lg">{formatCurrency(method === 'SLM' ? slmDep : wdvDep)}</span>
        </p>
      </div>

      <div className="overflow-x-auto">
        <table className="w-full text-xs border-collapse">
          <thead>
            <tr className="bg-gray-100">
              <th className="border p-1.5">Year</th>
              <th className="border p-1.5">Depreciation</th>
              <th className="border p-1.5">Book Value</th>
            </tr>
          </thead>
          <tbody>
            {rows.map((row) => (
              <tr key={row.year}>
                <td className="border p-1.5 text-center">{row.year}</td>
                <td className="border p-1.5 text-center">{formatCurrency(row.depreciation || slmDep)}</td>
                <td className="border p-1.5 text-center">{formatCurrency(row.bookValue)}</td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>

      <p className="text-xs text-gray-500 text-center">
        SLM: Equal depreciation every year. WDV: Fixed percentage on reducing balance.
      </p>
    </div>
  )
}
